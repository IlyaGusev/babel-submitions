\headerbox{Introduction}{name=introduction,column=0,row=0,span=3}{
\underline{\textbf{Data}}
\begin{itemize}
\item  2 monolingual corpora - around \textbf{1 million sentences} each
\item  Parallel corpus - \textbf{50000 sentences}
\item  Domain: \textbf{hotels} (probably Booking.com)
\item  Train language pair: \textbf{en-ru}
\item  Final test language pair is unknown, our guess: \textbf{en-nl}
\end{itemize}


\underline{\textbf{Testing system}}
\begin{itemize}
\item  Time limit - \textbf{8 hours}
\item  Disk space limit - \textbf{20 Gb}
\item  Error output is disabled in order to hide true test language pair
\end{itemize}
}

\headerbox{Language detection}{name=introduction,column=3,span=3}{
\begin{center}

\begin{itemize}
\item Python lib for language detection: lang\_detect
\item Language is not in the proposed list -> submition status is \textbf{SUCCESS}
\item Language is in the proposed list -> submition status is \textbf{ERROR}
\item Log(N) submitions for finding src or tgt language
\end{itemize}
}

%	MATERIALS AND METHODS
%----------------------------------------------------------------------------------------

\headerbox{UNMT}{name=methods1,column=0,below=introduction, span=3}{

\begin{center}
\includegraphics[scale=0.2]{model}
\end{center}

\begin{itemize}
\item Embeddings: MUSE-300
\item Encoder: bidirectional LSTM, 3 layers, 400 hidden size
\item Decoder: LSTM with attention (Luong, general), 3 layers, 400 hidden size
\item Discriminator: 3 fully connected layers, 1024 hidden size
\end{itemize}
}


\headerbox{Supervised NMT}{name=methods2,column=3,below=introduction, span=3}{
\begin{itemize}
\item Embeddings: fasttext-300
\item OpenNMT default RNN models
\item Encoder + Decoder from UNMT
\end{itemize}
}

\headerbox{Combined}{name=methods3,column=3,below=methods2, span=3}{
\begin{itemize}
\item N supervised epochs + N unsupervised batches
\item Joint training of unsupervised and supervised models
\end{itemize}
}

%----------------------------------------------------------------------------------------
%	RESULTS
%----------------------------------------------------------------------------------------
\headerbox{Results}{name=results2,span=3,column=0,below=methods1}{
%------------------------------------------------

\begin{itemize}
\item cat input.txt > output.txt is a very good baseline with BLEU around 2-3
\item Word by word models aren't good for en-ru language pair
\item Pure UNMT - can't converge in 8 hours
\item Supervised models - 50000 sentences is enough for 20-25 BLEU, mainly because of the strict domain of data
\item Always fix versions of the requirements. Gensim interface has changed in the last day of the competition
\end{itemize}

}
